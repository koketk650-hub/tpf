        function showToast(msg, isError = false) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-msg');
            const icon = toast.querySelector('i');

            toastMsg.innerText = msg;
            if(isError) {
                icon.className = 'fas fa-exclamation-circle';
                icon.style.color = 'var(--danger-red)';
            } else {
                icon.className = 'fas fa-check-circle';

            toast.classList.add('show');
            setTimeout(() => { toast.classList.remove('show'); }, 3000);
        }

        // --- UPGRADED: CoinCap First Priority Fetching ---
        async function updateLivePrices() {
            try {
                // 1. CoinCap (Global Neutral Primary Source - Executes First)
                const fetchCoinCap = fetch('https://api.coincap.io/v2/assets?limit=100')
                    .then(async r => {
                        if (!r.ok) throw new Error('CoinCap Blocked');
                        const data = await r.json();
                        const map = {};
                        data.data.forEach(item => { map[item.symbol] = parseFloat(item.priceUsd); });
                        return map;
                    });

                // Helper for Binance fallbacks
                const processBinance = async (url) => {
                    const res = await fetch(url);
                    if (!res.ok) throw new Error('API Blocked'); 
                    const data = await res.json();
                    const map = {};
                    data.forEach(item => { map[item.symbol.replace('USDT', '')] = parseFloat(item.price); });
                    return map;
                };

                // 2. Binance Global Fallback
                const fetchBinanceGlobal = processBinance('https://api.binance.com/api/v3/ticker/price');
                
                // 3. Binance US Fallback
                const fetchBinanceUS = processBinance('https://api.binance.us/api/v3/ticker/price');

                // RACE: CoinCap is given priority position in the Promise chain.
                // Whichever unblocked API returns valid data first will supply the UI.
                const priceMap = await Promise.any([
                    fetchCoinCap,
                    fetchBinanceGlobal, 
                    fetchBinanceUS 
                ]);

                coins.forEach(coin => {
                    if(coin.sym === 'USDC') return; // USDC remains at 1.00
                    if(priceMap[coin.sym]) {
                        coin.price = priceMap[coin.sym];
                    }
                });

                const activeCoin = coins.find(c => c.sym === currentSym);
                if(activeCoin && activeCoin.price > 0) {
                    currentPrice = activeCoin.price;
                    const isLow = currentPrice < 1;
                    const prec = isLow ? 6 : 2;
                    
                    document.getElementById('display-price-input').value = currentPrice.toFixed(prec);
                    document.getElementById('header-stats').innerText = `${currentPrice.toFixed(prec)}`;
                    
                    renderOrderBook();
                    updateCalculations();
                }
                renderCoinList();

            } catch (e) {
                console.log("Global price fetch blocked or delayed, retrying in next cycle...");
            }
        }

        async function fetchBalances() {
            const userId = localStorage.getItem('bitx_user_id');
            if(!userId) return;
            try {
                const res = await fetch(`${API_URL}/user/${userId}`);
                const data = await res.json();
                if(data.balance) {
                    serverBalances = data.balance;
                    updateBalancesUI();
                }
            } catch(e) {}
        }

            currentSym = sym;

            currentPrice = coin.price;
            document.getElementById('pair-title').innerText = `${sym}/USDT`;
            document.getElementById('avail-coin-label').innerText = `Avail ${sym}`;
            
            const isLow = currentPrice < 1;
            document.getElementById('display-price-input').value = currentPrice > 0 ? currentPrice.toFixed(isLow ? 6 : 2) : "Loading Price...";
            
            updateUIState();
            updateBalancesUI();
            toggleMenu(false);
            setSlider(0);
        }

        }

        function updateUIState() {
            const btn = document.getElementById('action-btn');
            const buyTab = document.getElementById('tab-buy');
            const sellTab = document.getElementById('tab-sell');
            const inp = document.getElementById('input-amount');
            const suffix = document.getElementById('input-unit');

            if(isBuy) {
                buyTab.className = 'bs-tab buy';
                sellTab.className = 'bs-tab sell';
                btn.style.background = 'var(--success-green)';
                btn.style.boxShadow = '0 4px 15px rgba(14, 203, 129, 0.2)';

                inp.placeholder = "Amount (USDT)";

                buyTab.className = 'bs-tab buy inactive';
                sellTab.className = 'bs-tab sell active-sell';

                btn.style.boxShadow = '0 4px 15px rgba(246, 70, 93, 0.2)';
                btn.innerText = `Sell ${currentSym}`;
                inp.placeholder = `Amount (${currentSym})`;

            }
            document.getElementById('input-amount').value = '';
            document.getElementById('est-result').innerText = isBuy ? `0.00 ${currentSym}` : `0.00 USDT`;
        }

        function updateBalancesUI() {

        }

        function updateCalculations() {
            const val = parseFloat(document.getElementById('input-amount').value) || 0;
            const resLabel = document.getElementById('est-result');
            
            if(currentPrice === 0 || isNaN(currentPrice)) {
                resLabel.innerText = "Waiting for price...";
                return;
            }

            if(isBuy) {
                const coinsReceived = val / currentPrice;
                resLabel.innerText = `${coinsReceived.toFixed(currentPrice < 1 ? 4 : 6)} ${currentSym}`;
            } else {
                const usdtReceived = val * currentPrice;
                resLabel.innerText = `${usdtReceived.toFixed(2)} USDT`;
            }
        }

        function setSlider(percent, e) {
            if(e) e.stopPropagation();
            document.getElementById('slider-thumb').style.left = percent + '%';
            document.getElementById('slider-val').innerText = percent + '%';

            const usdt = parseFloat(serverBalances['usdt'] || serverBalances['USDT'] || 0);
            const coin = parseFloat(serverBalances[currentSym.toLowerCase()] || serverBalances[currentSym] || 0);

            let maxInput = isBuy ? usdt : coin;

            const inputVal = maxInput * (percent / 100);
            const precision = isBuy ? 2 : (currentPrice < 1 ? 4 : 6);
            
            if (inputVal > 0) {
                document.getElementById('input-amount').value = inputVal.toFixed(precision);
            } else {
                document.getElementById('input-amount').value = '';
            }
            
            updateCalculations();
        }

        function sliderClick(e) {
            const rect = document.getElementById('slider-box').getBoundingClientRect();
            const x = e.clientX - rect.left;
            let percent = (x / rect.width) * 100;
            if(percent < 0) percent = 0;
            if(percent > 100) percent = 100;
            setSlider(percent);
        }

        async function executeTrade() {
            const userId = localStorage.getItem('bitx_user_id');
            const amountInput = parseFloat(document.getElementById('input-amount').value);

            if (!userId) return showToast("Please Login", true);
            if (!amountInput || amountInput <= 0) return showToast("Enter valid amount", true);
            if (currentPrice === 0) return showToast("Waiting for live price, try again.", true);

            let fromCoin, toCoin, sendAmount;

            if(isBuy) {
                fromCoin = 'USDT';
                toCoin = currentSym;
                sendAmount = amountInput; 
            } else {
                fromCoin = currentSym;
                toCoin = 'USDT';
                sendAmount = amountInput; 
            }

            const pFrom = isBuy ? 1 : currentPrice; 
            const pTo = isBuy ? currentPrice : 1;
            const rate = pFrom / pTo;

            const btn = document.getElementById('action-btn');
            btn.disabled = true;
            btn.innerText = "Processing...";

            try {
                const res = await fetch(`${API_URL}/exchange`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify({
                        userId,
                        fromCoin,
                        toCoin,
                        amount: sendAmount,
                        estimatedRate: rate
                    })
                });

                const data = await res.json();
                btn.disabled = false;
                
                if(res.ok) {
                    showToast("Order Successful!");
                    fetchBalances(); 
                    document.getElementById('input-amount').value = '';
                    setSlider(0);
                    btn.innerText = isBuy ? `Buy ${currentSym}` : `Sell ${currentSym}`;
                } else {
                    showToast("Trade Failed: " + (data.message || "Insufficient Balance"), true);
                    updateUIState();
                }
            } catch(e) { 
                showToast("Server Connection Error", true);
                btn.disabled = false;
                updateUIState();
            }
        }

        function renderOrderBook() {
            if(currentPrice === 0) return;
            const isLow = currentPrice < 1;
            const prec = isLow ? 6 : 2;
            
            const midEl = document.getElementById('mid-price');
            const prevPrice = parseFloat(midEl.innerText) || 0;
            
            midEl.innerText = currentPrice.toFixed(prec);
            midEl.className = currentPrice >= prevPrice ? "text-green" : "text-red";
            
            let asks = '';
            for(let i=5; i>=1; i--) {
                const p = (currentPrice + (i * (currentPrice*0.0002))).toFixed(prec);
                asks += `<div class="ob-row"><span class="ob-pos text-red">${i}</span><span class="ob-price text-red">${p}</span><span class="ob-qty">${(Math.random()*2).toFixed(4)}</span></div>`;
            }
            document.getElementById('asks-list').innerHTML = asks;

            let bids = '';
            for(let i=1; i<=5; i++) {
                const p = (currentPrice - (i * (currentPrice*0.0002))).toFixed(prec);
                bids += `<div class="ob-row"><span class="ob-pos text-green">${i}</span><span class="ob-price text-green">${p}</span><span class="ob-qty">${(Math.random()*2).toFixed(4)}</span></div>`;
            }
            document.getElementById('bids-list').innerHTML = bids;
        }

                </div>`;
            });
        }
