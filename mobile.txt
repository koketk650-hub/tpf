    <script>
        // --- NOTIFICATION LOGIC ---
        async function fetchNotifications() {
            const userId = localStorage.getItem('bitx_user_id');
            if (!userId) return;

            try {
                // Adjust to your actual API URL
                const response = await fetch(`/api/notifications/${userId}`); 
                if (!response.ok) return;
                const notifications = await response.json();
                
                const badge = document.getElementById('nav-badge');
                const list = document.getElementById('notification-list');
                
                const unread = notifications.filter(n => !n.isRead);
                
                if (unread.length > 0) {
                    badge.innerText = unread.length;
                    badge.style.display = 'block';
                } else {
                    badge.style.display = 'none';
                }

                if (notifications.length === 0) {
                    list.innerHTML = '<div class="nd-empty">No new notifications</div>';
                    return;
                }

                list.innerHTML = notifications.map(n => `
                    <div class="nd-item ${n.isRead ? '' : 'unread'}">
                        <div class="nd-title">${n.title}</div>
                        <div class="nd-msg">${n.message}</div>
                        <div style="font-size:10px; color:#666; margin-top:5px;">
                            ${new Date(n.createdAt).toLocaleString()}
                        </div>
                    </div>
                `).join('');
            } catch (error) {
                console.error("Failed to load notifications:", error);
            }
        }

        async function toggleNotifications() {
            const dropdown = document.getElementById('notification-dropdown');
            const isShowing = dropdown.style.display === 'flex';
            
            if (isShowing) {
                dropdown.style.display = 'none';
            } else {
                dropdown.style.display = 'flex';
                const userId = localStorage.getItem('bitx_user_id');
                if (userId) {
                    // Mark as read when opened
                    fetch(`/api/notifications/${userId}/read`, { method: 'POST' })
                        .then(() => fetchNotifications()); 
                }
            }
        }

        // Close dropdown when clicking outside
        document.addEventListener('click', function(e) {
            const wrapper = document.querySelector('.notification-wrapper');
            if (wrapper && !wrapper.contains(e.target)) {
                document.getElementById('notification-dropdown').style.display = 'none';
            }
        });

        // Initialize Notifications
        fetchNotifications();
        setInterval(fetchNotifications, 15000); // Check every 15 secs


        // --- SLIDESHOW LOGIC ---
        let currentSlide = 0;
        const track = document.getElementById('slides-track');
        const dots = document.querySelectorAll('.dot');
        const totalSlides = 3;

        function goToSlide(index) {
            currentSlide = index;
            const translateX = -(currentSlide * 33.333); // Move by 1/3 per slide
            track.style.transform = `translateX(${translateX}%)`;
            
            dots.forEach(d => d.classList.remove('active'));
            dots[currentSlide].classList.add('active');
        }

        setInterval(() => {
            let next = currentSlide + 1;
            if(next >= totalSlides) next = 0;
            goToSlide(next);
        }, 4000); 

        // --- MODAL FUNCTIONS ---
        function showModal(title, message, iconClass, primaryText, primaryCallback, secondaryText = null) {
            const overlay = document.getElementById('custom-modal');
            const content = overlay.querySelector('.modal-content');
            const actionsDiv = document.getElementById('modal-actions');
            
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-msg').innerText = message;
            document.getElementById('modal-icon').innerHTML = `<i class="fas ${iconClass}"></i>`;
            
            let buttonsHtml = '';
            if (secondaryText) {
                buttonsHtml += `<button class="modal-btn btn-secondary" onclick="closeModal()">${secondaryText}</button>`;
            }
            buttonsHtml += `<button class="modal-btn" id="btn-primary">${primaryText}</button>`;
            actionsDiv.innerHTML = buttonsHtml;

            document.getElementById('btn-primary').onclick = function() {
                closeModal();
                if(primaryCallback) setTimeout(primaryCallback, 300);
            };

            overlay.style.display = 'flex';
            setTimeout(() => {
                overlay.style.opacity = '1';
                content.style.transform = 'scale(1)';
            }, 10);
        }

        function closeModal() {
            const overlay = document.getElementById('custom-modal');
            const content = overlay.querySelector('.modal-content');
            overlay.style.opacity = '0';
            content.style.transform = 'scale(0.8)';
            setTimeout(() => { overlay.style.display = 'none'; }, 300);
        }

        // --- DATA FETCHING (PRESERVED) ---
        let globalMarketData = [];
        
        async function fetchHomeData() {
            try {
                const response = await fetch('https://api.coingecko.com/api/v3/coins/markets?vs_currency=usd&order=market_cap_desc&per_page=10&page=1&sparkline=false');
                globalMarketData = await response.json();
                renderUI();
            } catch (error) {
                console.error("Data Fetch Error:", error);
                document.getElementById('home-market-list').innerHTML = '<div style="padding:20px; text-align:center; color: var(--text-muted);">Failed to load data</div>';
            }
        }

        function simulateTicks() {
            if(globalMarketData.length === 0) return;
            globalMarketData.forEach(coin => {
                coin.current_price *= (1 + (Math.random() * 0.001 - 0.0005));
                coin.price_change_percentage_24h += (Math.random() - 0.5) * 0.02;
            });
            renderUI();
        }

        function renderUI() {
            const tickerContainer = document.getElementById('ticker-container');
            let tickerHtml = '';
            globalMarketData.slice(0, 3).forEach(coin => {
                const change = coin.price_change_percentage_24h.toFixed(2);
                const colorClass = change >= 0 ? 't-green' : 't-red';
                tickerHtml += `
                <div class="ht-item" onclick="window.location.href='mobile_exchange.html?symbol=${coin.symbol.toUpperCase()}'">
                    <div class="ht-head"><img src="${coin.image}" width="14"> ${coin.symbol.toUpperCase()}/USDT</div>
                    <span class="ht-price">${coin.current_price.toLocaleString(undefined, {minimumFractionDigits: 2})}</span>
                    <span class="${colorClass}">${change >= 0 ? '+' : ''}${change}%</span>
                </div>`;
            });
            tickerContainer.innerHTML = tickerHtml;

            const listContainer = document.getElementById('home-market-list');
            let listHtml = '';
            globalMarketData.forEach(coin => {
                const change = coin.price_change_percentage_24h.toFixed(2);
                const colorClass = change >= 0 ? 'bg-green' : 'bg-red';
                listHtml += `
                <div class="mm-row" onclick="window.location.href='mobile_exchange.html?symbol=${coin.symbol.toUpperCase()}'">
                    <div class="mm-left">
                        <img src="${coin.image}" class="mm-icon">
                        <div class="mm-name"><h4>${coin.symbol.toUpperCase()}</h4><span>/USDT</span></div>
                    </div>
                    <div class="mm-price">${coin.current_price.toLocaleString(undefined, {minimumFractionDigits: 2})}</div>
                    <div class="mm-btn ${colorClass}">${change >= 0 ? '+' : ''}${change}%</div>
                </div>`;
            });
            listContainer.innerHTML = listHtml;
        }

        fetchHomeData(); 
        setInterval(fetchHomeData, 60000); 
        setInterval(simulateTicks, 1000);
