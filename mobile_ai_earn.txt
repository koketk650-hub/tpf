        // Configuration for Coin Images
        const coinImages = {

        async function fetchLiveEarnData() {
            try {
                // 1. Fetch Real Market Prices from your Server
                const res = await fetch('/api/market-data'); 
                const data = await res.json();
                
                const container = document.getElementById('earn-list');
                container.innerHTML = '';

                // 2. Always add USDT (Stable)
                addCard(container, 'USDT', 'Smart Arbitrage', '12.50%', '7 Days', true);

                // 3. Filter for major coins and generate dynamic cards
                const targetCoins = ['BTC', 'ETH', 'SOL', 'BNB', 'ADA'];
                
                targetCoins.forEach(sym => {
                    const coinData = data.find(c => c.symbol === `${sym}USDT`);
                    if(coinData) {
                        // Generate a "Real-looking" APR based on price hash
                        const price = parseFloat(coinData.price);
                        // Logic: Higher price often correlates to lower volatility yield, but we fake it for demo
                        const dynamicAPR = ((price % 15) + 3).toFixed(2) + '%'; 
                        
                        addCard(container, sym, 'Trend Grid', dynamicAPR, 'Flexible', false);
                    }
                });

            } catch (e) {
                console.error("Error loading earn data", e);
                document.getElementById('earn-list').innerHTML = '<div style="text-align:center; color:#555;">Market data unavailable</div>';
            }
        }

        function addCard(container, symbol, strategyName, apr, lock, isHot) {
            const img = coinImages[symbol] || coinImages['BTC'];
            const tagHtml = isHot ? '<div class="tag hot">HOT</div>' : '<div class="tag">AI AUTO</div>';
            
            const html = `
            <div class="prod-card">
                ${tagHtml}
                <div class="pc-head">
                    <img src="${img}" class="pc-icon"> 
                    <span class="pc-title">${symbol} ${strategyName}</span>
                </div>
                <div class="pc-stats">
                    <div class="stat-box">
                        <h4>${apr}</h4>
                        <p>Est. APR</p>
                    </div>
                    <div class="stat-box" style="text-align:right">
                        <h4 style="color:#fff">${lock}</h4>
                        <p>Lock Period</p>
                    </div>
                </div>
                <button class="btn-sub" onclick="subscribe('${symbol} ${strategyName}')">Subscribe</button>
            </div>`;
            
            container.innerHTML += html;
        }

        function subscribe(name) {
            const t = document.getElementById('toast');
            t.innerText = `Subscribed to ${name}`;
            t.style.display = 'block';
            
            // Optional: You could add a fetch call here to save subscription to server
            // fetch('/api/earn/subscribe', { method: 'POST', ... })

            setTimeout(() => t.style.display = 'none', 2000);
        }

        function animateTVL() {
            const el = document.getElementById('tvl-display');
            let val = 12450293;
            
            setInterval(() => {
                // Simulate live updates
                const change = Math.floor(Math.random() * 500) - 200; // Random + or -
                val += change;
                el.innerText = '$' + val.toLocaleString();
                
                // Color flash effect
                el.style.color = change > 0 ? 'var(--green)' : (change < 0 ? 'var(--primary)' : 'var(--primary)');
                setTimeout(() => el.style.color = 'var(--primary)', 500);
            }, 3000);
        }
