        // 3. SELECT COIN LOGIC
        function selectCoin(sym) {
            const coin = coins.find(c => c.sym === sym);
            currentCoinSym = sym;
            
            // Update Icons & Text
            document.getElementById('sel-sym').innerText = sym;
            document.getElementById('sel-icon').src = coin.img;
            document.getElementById('unit-display').innerText = sym;
            document.getElementById('avail-unit').innerText = sym;

            // Update Balance
            currentBal = parseFloat(localStorage.getItem(`bitx_${sym.toLowerCase()}`) || 0);
            document.getElementById('avail-bal').innerText = currentBal.toFixed(4);

            toggleMenu(false);
        }

        // 4. SET MAX
        function setMax() {
            document.getElementById('amount-input').value = currentBal;
        }

        // 5. RESTRICTION LOGIC
function triggerRestriction() {
    const amount = document.getElementById('amount-input').value;
    
    if(!amount || amount <= 0) {
        alert("Please enter a valid amount.");
        return;
    }

    // --- NEW: RECORD FAILED TRANSFER ---
    const now = new Date();
    const timeString = now.toISOString().slice(0, 19).replace('T', ' ');

    const newRecord = {
        type: 'transfer',
        coin: 'USDT', // Currently hardcoded to USDT in your UI
        amount: amount,
        time: timeString,
        status: 'failed' // Automatically marked failed
    };

    const history = JSON.parse(localStorage.getItem('bitx_history') || '[]');
    history.push(newRecord);
    localStorage.setItem('bitx_history', JSON.stringify(history));
    // ----------------------------------

    alert("Transfer currently not allowed between spot trade and Futures trade.");
}

        // 6. MENU UTILS
        function renderCoinList() {
            const list = document.getElementById('coin-list');
            list.innerHTML = "";
            coins.forEach(c => {
                list.innerHTML += `
                <div class="bm-item" onclick="selectCoin('${c.sym}')">
                    <img src="${c.img}" class="bm-icon">
                    <div style="font-weight:600">${c.sym}</div>
                </div>`;
            });
        }

        function toggleMenu(show) {
            const overlay = document.getElementById('overlay');
            const menu = document.getElementById('coin-menu');
            if(show) {
                overlay.style.display = 'block';
                menu.classList.add('active');
            } else {
                overlay.style.display = 'none';
                menu.classList.remove('active');
            }
        }

        // 7. ANIMATION
        function rotateIcon(el) {
            el.style.transform = "translateY(-50%) rotate(180deg)";
