    <script>
        // --- TOAST HELPER ---
        function showToast(msg, isError = false) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-msg');
            const icon = toast.querySelector('i');

            toastMsg.innerText = msg;
            if(isError) {
                icon.className = 'fas fa-exclamation-circle';
                icon.style.color = 'var(--danger-red)';
            } else {
                icon.className = 'fas fa-check-circle';
                icon.style.color = 'var(--success-green)';
            }

            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }

        // --- AUTH CHECK ---
        const userId = localStorage.getItem('bitx_user_id');
        if (!userId) {
            setTimeout(() => {
                showToast("Please Login to view assets", true);
                setTimeout(() => window.location.href = 'login.html', 1500);
            }, 500);
        }

        // --- NOTIFICATION LOGIC ---
        async function fetchNotifications() {
            if (!userId) return;

            try {
                const response = await fetch(`/api/notifications/${userId}`); 
                if (!response.ok) return;
                const notifications = await response.json();
                
                const badge = document.getElementById('nav-badge');
                const list = document.getElementById('notification-list');
                
                const unread = notifications.filter(n => !n.isRead);
                
                if (unread.length > 0) {
                    badge.innerText = unread.length;
                    badge.style.display = 'block';
                } else {
                    badge.style.display = 'none';
                }

                if (notifications.length === 0) {
                    list.innerHTML = '<div class="nd-empty">No new notifications</div>';
                    return;
                }

                list.innerHTML = notifications.map(n => `
                    <div class="nd-item ${n.isRead ? '' : 'unread'}">
                        <div class="nd-title">${n.title}</div>
                        <div class="nd-msg">${n.message}</div>
                        <div style="font-size:10px; color:var(--text-muted); margin-top:5px;">
                            ${new Date(n.createdAt).toLocaleString()}
                        </div>
                    </div>
                `).join('');
            } catch (error) {
                console.error("Failed to load notifications:", error);
            }
        }

        async function toggleNotifications() {
            const dropdown = document.getElementById('notification-dropdown');
            const isShowing = dropdown.style.display === 'flex';
            
            if (isShowing) {
                dropdown.style.display = 'none';
            } else {
                dropdown.style.display = 'flex';
                if (userId) {
                    fetch(`/api/notifications/${userId}/read`, { method: 'POST' })
                        .then(() => fetchNotifications()); 
                }
            }
        }

        document.addEventListener('click', function(e) {
            const wrapper = document.querySelector('.notification-wrapper');
            if (wrapper && !wrapper.contains(e.target)) {
                document.getElementById('notification-dropdown').style.display = 'none';
            }
        });

        // --- ICON MAPPER ---
        function getIcon(symbol) {
            const icons = {
                'BTC': 'https://assets.coingecko.com/coins/images/1/small/bitcoin.png',
                'ETH': 'https://assets.coingecko.com/coins/images/279/small/ethereum.png',
                'USDT': 'https://assets.coingecko.com/coins/images/325/small/Tether.png',
                'BNB': 'https://assets.coingecko.com/coins/images/825/small/binance-coin-logo.png',
                'SOL': 'https://assets.coingecko.com/coins/images/4128/small/solana.png',
                'XRP': 'https://assets.coingecko.com/coins/images/44/small/xrp-symbol-white-128.png',
                'USDC': 'https://assets.coingecko.com/coins/images/6319/small/usdc.png',
                'ADA': 'https://assets.coingecko.com/coins/images/975/small/cardano.png',
                'DOGE': 'https://assets.coingecko.com/coins/images/5/small/dogecoin.png',
                'TRX': 'https://assets.coingecko.com/coins/images/1094/small/tron-logo.png',
                'AVAX': 'https://assets.coingecko.com/coins/images/12559/small/Avalanche_Circle_RedWhite_Trans.png',
                'DOT': 'https://assets.coingecko.com/coins/images/12171/small/polkadot.png',
                'LTC': 'https://assets.coingecko.com/coins/images/2/small/litecoin.png',
                'SHIB': 'https://assets.coingecko.com/coins/images/11939/small/shiba.png'
            };
            return icons[symbol] || 'https://assets.coingecko.com/coins/images/1/small/bitcoin.png';
        }

        // --- STATE ---
        let userBalances = {};
        let marketPrices = {}; 
        let totalValUSDT = 0;
        let isBalanceHidden = false;

        async function fetchAssets() {
            if(!userId) return;
            try {
                // 1. GET USER BALANCE
                const userRes = await fetch(`/api/user/${userId}`);
                if (!userRes.ok) throw new Error("Failed to fetch user");
                const userData = await userRes.json();

                if (userData.balance) {
                    userBalances = {
                        'USDT': userData.balance.usdt || 0,
                        'BTC': userData.balance.btc || 0,
                        'ETH': userData.balance.eth || 0,
                        'BNB': userData.balance.bnb || 0,
                        'SOL': userData.balance.sol || 0,
                        'XRP': userData.balance.xrp || 0,
                        'USDC': userData.balance.usdc || 0,
                        'ADA': userData.balance.ada || 0,
                        'DOGE': userData.balance.doge || 0,
                        'TRX': userData.balance.trx || 0,
                        'AVAX': userData.balance.avax || 0,
                        'SHIB': userData.balance.shib || 0,
                        'DOT': userData.balance.dot || 0,
                        'LTC': userData.balance.ltc || 0
                    };
                }

                // 2. GET MARKET PRICES
                const priceRes = await fetch('/api/market-data');
                if (!priceRes.ok) throw new Error("Market API Error");
                
                const priceData = await priceRes.json();
                
                // Store prices
                marketPrices['USDT'] = 1.0; 
                const cleanCoins = priceData.map(item => {
                    let symbol = item.symbol.replace('USDT', '');
                    let price = parseFloat(item.price);
                    marketPrices[symbol] = price; 
                    
                    return {
                        symbol: symbol,
                        current_price: price,
                        image: getIcon(symbol)
                    };
                });
                
                calculateTotalValue(cleanCoins);
                renderAssetList(cleanCoins);

            } catch (e) {
                console.error("Asset Error", e);
                document.getElementById('asset-list-container').innerHTML = 
                    `<div style="text-align:center; padding:20px; color:var(--danger-red);">Connection Failed</div>`;
                showToast("Failed to load assets", true);
            }
        }
        
        function calculateTotalValue(coins) {
            let total = userBalances['USDT'] || 0;
            coins.forEach(coin => {
                const sym = coin.symbol;
                if (sym === 'USDT') return;
                const balance = userBalances[sym] || 0;
                total += balance * coin.current_price;
            });
            totalValUSDT = total;
            updateTotalDisplay();
        }

        function updateTotalDisplay() {
            if (isBalanceHidden) {
                document.getElementById('total-amount-display').innerText = "******";
                document.getElementById('equiv-display').innerText = "******";
                return;
            }

            const selectedCurrency = document.getElementById('base-currency').value;
            const rate = marketPrices[selectedCurrency] || 1;
            
            // Convert Total USDT Value to Selected Currency
            const convertedTotal = totalValUSDT / rate;
            
            let displayTotal = "";
            let equivText = "";

            if(selectedCurrency === 'USDT') {
                displayTotal = convertedTotal.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2});
                equivText = `≈ $${displayTotal}`;
            } else {
                displayTotal = convertedTotal.toFixed(8);
                equivText = `≈ $${totalValUSDT.toLocaleString('en-US', {minimumFractionDigits: 2, maximumFractionDigits: 2})}`;
            }

            document.getElementById('total-amount-display').innerText = displayTotal;
            document.getElementById('equiv-display').innerText = equivText;
        }

        function renderAssetList(coins) {
            const container = document.getElementById('asset-list-container');
            let html = '';
            
            // Add USDT manually
            const usdtBal = userBalances['USDT'] || 0;
            const usdtVal = usdtBal * 1.00;
            html += createAssetRow('USDT', 'Tether', usdtBal, usdtVal, getIcon('USDT'));

            coins.forEach(coin => {
                const sym = coin.symbol;
                if (sym === 'USDT') return;

                const balance = userBalances[sym] || 0;
                const value = balance * coin.current_price;
                
                html += createAssetRow(sym, sym, balance, value, coin.image);
            });
            container.innerHTML = html;
            
            toggleZeroBalances();
            if(isBalanceHidden) applyMaskToRows(true);
        }

        function createAssetRow(sym, name, balance, value, img) {
            const rowClass = balance === 0 ? 'asset-row zero-balance' : 'asset-row';
            const displayBal = balance < 0.000001 ? "0.00" : balance.toFixed(6);
            
            return `
            <div class="${rowClass}">
                <div class="ar-left">
                    <img src="${img}" class="ar-icon" alt="${sym}">
                    <div class="ar-name"><h4>${sym}</h4><span>${name}</span></div>
                </div>
                <div class="ar-right">
                    <span class="ar-amount asset-val-hidden">${displayBal}</span>
                    <span class="ar-fiat asset-val-hidden">≈ $${value.toFixed(2)}</span>
                </div>
            </div>`;
        }

        function toggleTotalBalance() {
            isBalanceHidden = !isBalanceHidden;
            const eye = document.getElementById('eye-icon');
            
            if (isBalanceHidden) {
                eye.className = "fas fa-eye-slash";
                document.getElementById('total-amount-display').innerText = "******";
                document.getElementById('equiv-display').innerText = "******";
                applyMaskToRows(true);
            } else {
                eye.className = "fas fa-eye";
                updateTotalDisplay();
                applyMaskToRows(false);
            }
        }

        function applyMaskToRows(hide) {
            const amounts = document.querySelectorAll('.ar-amount');
            const fiats = document.querySelectorAll('.ar-fiat');
            
            if(hide) {
                amounts.forEach(el => el.innerText = "*****");
                fiats.forEach(el => el.innerText = "*****");
            } else {
                fetchAssets();
            }
        }

        function toggleZeroBalances() {
            const checkBox = document.getElementById('hide-zero-check');
            const zeroRows = document.querySelectorAll('.zero-balance');
            zeroRows.forEach(row => row.style.display = checkBox.checked ? 'none' : 'flex');
        }

        // Start
        window.onload = () => {
            fetchNotifications();
            setInterval(fetchNotifications, 15000);
            fetchAssets();
        }
    </script>
