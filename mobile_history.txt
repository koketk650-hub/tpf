    <script>
    const API_URL = '/api';
    let allTxns = [];

    window.onload = () => {
        // 1. Load Cached Data Immediately (Instant Feel)
        const cached = localStorage.getItem('bitx_history_cache');
        if (cached) {
            try {
                allTxns = JSON.parse(cached);
                renderList(allTxns);
            } catch(e) { console.log("Cache invalid"); }
        }
        
        // 2. Fetch Fresh Data (With timestamp to break cache)
        loadHistory();
    };

    async function loadHistory() {
        const userId = localStorage.getItem('bitx_user_id');
        if(!userId) return window.location.href = 'login.html';

        try {
            // 🔥 FIXED: Added timestamp `?t=` to prevent caching
            const res = await fetch(`${API_URL}/history/${userId}?t=${new Date().getTime()}`);
            const newData = await res.json();
            
            // Render if new data exists
            if (JSON.stringify(newData) !== JSON.stringify(allTxns)) {
                allTxns = newData;
                localStorage.setItem('bitx_history_cache', JSON.stringify(allTxns)); 
                renderList(allTxns);
            }
        } catch(e) {
            console.error(e);
            if(allTxns.length === 0) {
                document.getElementById('history-list').innerHTML = '<div class="empty-msg">Failed to load history</div>';
            }
        }
    }

    function renderList(data) {
        const container = document.getElementById('history-list');
        
        if(!data || data.length === 0) {
            container.innerHTML = '<div class="empty-msg">No transactions found</div>';
            return;
        }

        const listHTML = data.map(txn => {
            // 🔥 FIXED: Handle missing types to prevent crashing
            const type = txn.type ? txn.type.toLowerCase() : 'unknown';
            const status = txn.status ? txn.status.toLowerCase() : 'pending';
            const amount = parseFloat(txn.amount || 0);

            let iconClass = 'icon-trade';
            let iconSymbol = 'fa-exchange-alt';
            let amtClass = 'amt-pos';
            let amtPrefix = '+';
            let statusClass = 'st-success';
            let title = type.toUpperCase();

            // LOGIC TO DETERMINE STYLES
            if (type === 'deposit') {
                iconClass = 'icon-deposit';
                iconSymbol = 'fa-arrow-down';
                title = `Deposit ${txn.coin || ''}`;
                if(status === 'pending') statusClass = 'st-pending';
                else if(status === 'rejected') { statusClass = 'st-rejected'; amtClass = 'amt-neg'; }
                else statusClass = 'st-success';
            } 
            else if (type === 'withdraw') {
                iconClass = status === 'pending' ? 'icon-pending' : 'icon-withdraw';
                iconSymbol = 'fa-arrow-up';
                amtClass = 'amt-neg';
                amtPrefix = '-';
                title = `Withdraw ${txn.coin || ''}`;
                if(status === 'pending') statusClass = 'st-pending';
                else if(status === 'rejected') statusClass = 'st-rejected';
                else statusClass = 'st-success';
            }
            else if (type === 'trade') {
                if(status === 'win') {
                    iconClass = 'icon-deposit';
                    iconSymbol = 'fa-trophy';
                    title = 'Trade Profit';
                    statusClass = 'st-win';
                    amtPrefix = '+';
                } else {
                    iconClass = 'icon-withdraw';
                    iconSymbol = 'fa-times';
                    amtClass = 'amt-neg';
                    amtPrefix = '-';
                    title = 'Trade Loss';
                    statusClass = 'st-loss';
                }
            }
            else if (type === 'spot') {
                iconClass = 'icon-trade';
                iconSymbol = 'fa-sync-alt';
                amtClass = 'amt-neg'; 
                amtPrefix = '';
                title = txn.description || 'Spot Exchange';
                statusClass = 'st-success';
            }
            else {
                // Handle unknown/ghost types safely
                iconClass = 'icon-unknown';
                iconSymbol = 'fa-question';
                title = 'Unknown Transaction';
                statusClass = 'st-pending';
            }

            const dateObj = new Date(txn.date);
            const dateStr = dateObj.toLocaleDateString() + ' ' + dateObj.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});

            return `
            <div class="txn-card">
                <div class="txn-left">
                    <div class="txn-icon ${iconClass}"><i class="fas ${iconSymbol}"></i></div>
                    <div class="txn-info">
                        <h4>${title}</h4>
                        <p>${dateStr}</p>
                    </div>
                </div>
                <div class="txn-right">
                    <span class="txn-amount ${amtClass}">${amtPrefix}${amount.toFixed(4)} ${txn.coin || 'USDT'}</span>
                    <span class="status-badge ${statusClass}">${status.toUpperCase()}</span>
                </div>
            </div>`;
        }).join('');

        container.innerHTML = listHTML;
    }

    function filterType(type, el) {
        document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
        el.classList.add('active');

        if(type === 'all') {
            renderList(allTxns);
        } else {
            // Safe filter that checks if type exists
            const filtered = allTxns.filter(t => t.type && t.type.toLowerCase() === type);
            renderList(filtered);
        }
    }
</script>
