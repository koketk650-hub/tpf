    <script>
        const API_URL = '/api';

        // --- TOAST HELPER ---
        function showToast(msg, isError = false) {
            const toast = document.getElementById('toast');
            const toastMsg = document.getElementById('toast-msg');
            const icon = toast.querySelector('i');

            toastMsg.innerText = msg;
            if(isError) {
                icon.className = 'fas fa-exclamation-circle';
                icon.style.color = 'var(--danger-red)';
            } else {
                icon.className = 'fas fa-check-circle';
                icon.style.color = 'var(--success-green)';
            }

            toast.classList.add('show');
            setTimeout(() => {
                toast.classList.remove('show');
            }, 3000);
        }

        async function sendCode() {
            const email = document.getElementById('email').value;
            if(!email) return showToast("Please enter email", true);

            const btn = document.getElementById('btn-get-code');
            btn.disabled = true;
            btn.innerText = "Sending...";

            try {
                const res = await fetch(`${API_URL}/forgot-password`, {
                    method: 'POST', headers:{'Content-Type':'application/json'},
                    body: JSON.stringify({ email })
                });
                const data = await res.json();
                
                if(res.ok) {
                    showToast("Code Sent! Check your email");
                    // SIMULATION: In a real app, this goes to email.
                    console.log("Simulated Code:", data.code || '123456');
                    startTimer(60);
                } else {
                    showToast(data.message, true);
                    btn.disabled = false;
                    btn.innerText = "Get Code";
                }
            } catch(e) { 
                showToast("Connection Error", true); 
                btn.disabled = false; 
                btn.innerText = "Get Code";
            }
        }

        function startTimer(sec) {
            const btn = document.getElementById('btn-get-code');
            let t = sec;
            const timer = setInterval(() => {
                btn.innerText = `${t}s`;
                t--;
                if(t < 0) {
                    clearInterval(timer);
                    btn.disabled = false;
                    btn.innerText = "Get Code";
                }
            }, 1000);
        }

        async function resetPassword() {
            const email = document.getElementById('email').value;
            const code = document.getElementById('code').value;
            const newPass = document.getElementById('new-pass').value;

            if(!email || !code || !newPass) return showToast("Please fill all fields", true);
            if(newPass.length < 6) return showToast("Password too short (min 6 chars)", true);

            try {
                const res = await fetch(`${API_URL}/reset-password`, {
                    method: 'POST', headers:{'Content-Type':'application/json'},
                    body: JSON.stringify({ email, code, newPass })
                });
                if(res.ok) {
                    showToast("Password Reset Successful!");
                    setTimeout(() => {
                        window.location.href = 'login.html';
                    }, 1500);
                } else {
                    showToast("Failed: Invalid code or email", true);
                }
            } catch(e) { showToast("Connection Error", true); }
        }
