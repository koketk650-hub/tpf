    <script>
        const API_URL = '/api';
        let images = { front: '', back: '', selfie: '' };
        let streamRef = null;

        window.onload = async function() {
            const userId = localStorage.getItem('bitx_user_id');
            if(!userId) return window.location.href = 'login.html';

            try {
                const res = await fetch(`${API_URL}/user/${userId}`);
                const user = await res.json();
                const status = user.kyc ? user.kyc.status : 'unverified';

                if(status === 'verified') {
                    document.getElementById('kyc-form').style.display = 'none';
                    document.getElementById('status-verified').style.display = 'block';
                } else if (status === 'pending') {
                    document.getElementById('kyc-form').style.display = 'none';
                    document.getElementById('status-pending').style.display = 'block';
                }
            } catch(e) { console.error(e); }
        };

        // UI HELPERS (MODALS RESTORED)
        function showModal(title, message, isError = true) {
            const overlay = document.getElementById('custom-modal');
            const icon = document.getElementById('modal-icon');
            
            document.getElementById('modal-title').innerText = title;
            document.getElementById('modal-msg').innerText = message;
            
            if(isError) {
                icon.innerHTML = '<i class="fas fa-times-circle"></i>';
                icon.style.color = 'var(--danger-red)';
            } else {
                icon.innerHTML = '<i class="fas fa-check-circle"></i>';
                icon.style.color = 'var(--success-green)';
            }

            overlay.style.display = 'flex';
            setTimeout(() => {
                overlay.style.opacity = '1';
                document.querySelector('.modal-content').style.transform = 'scale(1)';
            }, 10);
        }

        function closeModal() {
            const overlay = document.getElementById('custom-modal');
            overlay.style.opacity = '0';
            document.querySelector('.modal-content').style.transform = 'scale(0.8)';
            setTimeout(() => { overlay.style.display = 'none'; }, 300);
        }

        function toggleOtherId() {
            document.getElementById('other-id-group').style.display = 
                (document.getElementById('inp-type').value === 'Other') ? 'block' : 'none';
        }

        // STANDARD FILE HANDLING
        function handleFile(input, type) {
            if (input.files && input.files[0]) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    setPreviewData(type, e.target.result);
                };
                reader.readAsDataURL(input.files[0]);
            }
        }

        function setPreviewData(type, base64Data) {
            const img = document.getElementById(`prev-${type}`);
            const clearBtn = document.getElementById(`clear-${type}`);
            
            img.src = base64Data;
            img.style.display = 'block';
            clearBtn.style.display = 'flex'; 
            images[type] = base64Data;
        }

        function clearFile(event, type) {
            event.stopPropagation(); 
            document.getElementById(`file-${type}`).value = ""; 
            document.getElementById(`prev-${type}`).style.display = "none"; 
            document.getElementById(`clear-${type}`).style.display = "none"; 
            images[type] = ""; 
        }

        // --- LIVE FACE SCANNER LOGIC ---
        async function startFaceScan() {
            if(images.selfie !== '') return;

            const scannerOverlay = document.getElementById('scanner-overlay');
            const video = document.getElementById('camera-feed');
            const laser = document.getElementById('scan-laser');
            const btn = document.getElementById('btn-capture');
            const text = document.getElementById('scanner-text');

            scannerOverlay.style.display = 'flex';
            text.innerHTML = "Initializing Camera...<br><span style='font-size: 12px; color: var(--text-muted);'>(Please allow camera access)</span>";
            laser.style.display = 'none';
            btn.style.display = 'none';

            try {
                const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "user" }, audio: false });
                streamRef = stream;
                video.srcObject = stream;
                
                video.onloadedmetadata = () => {
                    video.play().catch(e => console.error("Play error:", e));
                    text.innerText = "Aligning face...";
                    setTimeout(() => {
                        text.innerText = "Put your face into circle and click button below.";
                        laser.style.display = 'block';
                        btn.style.display = 'block';
                    }, 2000);
                };
            } catch (err) {
                console.warn("Camera access denied or unsupported.", err);
                stopFaceScan();
                document.getElementById('file-selfie').click();
            }
        }

        function stopFaceScan() {
            if(streamRef) {
                streamRef.getTracks().forEach(track => track.stop());
                streamRef = null;
            }
            document.getElementById('scanner-overlay').style.display = 'none';
        }

        function captureFaceScan() {
            const video = document.getElementById('camera-feed');
            const canvas = document.getElementById('canvas-capture');
            const context = canvas.getContext('2d');
            const text = document.getElementById('scanner-text');

            text.innerText = "Processing...";
            canvas.width = video.videoWidth;
            canvas.height = video.videoHeight;
            context.drawImage(video, 0, 0, canvas.width, canvas.height);
            context.translate(canvas.width, 0);
            context.scale(-1, 1);
            context.drawImage(video, 0, 0, canvas.width, canvas.height);
            
            const dataUrl = canvas.toDataURL('image/jpeg', 0.9);
            setPreviewData('selfie', dataUrl);
            
            setTimeout(() => { stopFaceScan(); }, 300);
        }

        // SUBMISSION
        async function submitKYC() {
            const name = document.getElementById('inp-name').value;
            const country = document.getElementById('inp-country').value;
            const type = document.getElementById('inp-type').value === 'Other' ? document.getElementById('inp-other-type').value : document.getElementById('inp-type').value;
            const number = document.getElementById('inp-number').value;

            if(!name || !country || !number || !images.front || !images.back || !images.selfie) {
                showModal("Incomplete", "Please complete all fields and ensure all three photos are uploaded.", true);
                return;
            }

            document.getElementById('loading-overlay').style.display = 'flex';
            
            const payload = { 
                userId: localStorage.getItem('bitx_user_id'),
                name, country, idType: type, idNumber: number,
                frontImg: images.front, backImg: images.back, selfieImg: images.selfie
            };

            try {
                const res = await fetch(`${API_URL}/kyc/submit`, {
                    method: 'POST',
                    headers: {'Content-Type': 'application/json'},
                    body: JSON.stringify(payload)
                });
                
                if(res.ok) {
                    showModal("Success", "Submission successful! Your documents are under review.", false);
                    setTimeout(() => window.location.reload(), 2000);
                } else {
                    showModal("Submission Failed", "There was an issue submitting your documents. Please try again.", true);
                }
            } catch(e) {
                showModal("Connection Error", "Error connecting to server. Please try again later.", true);
            } finally {
                document.getElementById('loading-overlay').style.display = 'none';
            }
        }
    </script>
