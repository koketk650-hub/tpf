        // --- NOTIFICATION LOGIC ---
        async function fetchNotifications() {
            const userId = localStorage.getItem('bitx_user_id');
            if (!userId) return;

            try {
                const response = await fetch(`/api/notifications/${userId}`); 
                if (!response.ok) return;
                const notifications = await response.json();
                
                const badge = document.getElementById('nav-badge');
                const list = document.getElementById('notification-list');
                
                const unread = notifications.filter(n => !n.isRead);
                
                if (unread.length > 0) {
                    badge.innerText = unread.length;
                    badge.style.display = 'block';
                } else {
                    badge.style.display = 'none';
                }

                if (notifications.length === 0) {
                    list.innerHTML = '<div class="nd-empty">No new notifications</div>';
                    return;
                }

                list.innerHTML = notifications.map(n => `
                    <div class="nd-item ${n.isRead ? '' : 'unread'}">
                        <div class="nd-title">${n.title}</div>
                        <div class="nd-msg">${n.message}</div>
                        <div style="font-size:10px; color:var(--text-muted); margin-top:5px;">
                            ${new Date(n.createdAt).toLocaleString()}
                        </div>
                    </div>
                `).join('');
            } catch (error) {
                console.error("Failed to load notifications:", error);
            }
        }

        async function toggleNotifications() {
            const dropdown = document.getElementById('notification-dropdown');
            const isShowing = dropdown.style.display === 'flex';
            
            if (isShowing) {
                dropdown.style.display = 'none';
            } else {
                dropdown.style.display = 'flex';
                const userId = localStorage.getItem('bitx_user_id');
                if (userId) {
                    fetch(`/api/notifications/${userId}/read`, { method: 'POST' })
                        .then(() => fetchNotifications()); 
                }
            }
        }

        async function raceForPrice() {
            const fetchCoinbase = fetch(`https://api.coinbase.com/v2/prices/${currentSym}-USD/spot`)
                .then(r => r.json())
                .then(d => ({ source: 'coinbase', price: parseFloat(d.data.amount) }));

            const fetchCoinCap = fetch(`https://api.coincap.io/v2/assets/${currentName}`)
                .then(r => r.json())
                .then(d => ({ source: 'coincap', price: parseFloat(d.data.priceUsd) }));

            const fetchBinance = fetch(`https://api.binance.com/api/v3/ticker/price?symbol=${currentSym}USDT`)
                .then(r => r.json())
                .then(d => ({ source: 'binance', price: parseFloat(d.price) }));

            try {
                const winner = await Promise.any([fetchCoinbase, fetchCoinCap, fetchBinance]);
                apiMode = winner.source;
                updatePriceUI(winner.price);
            } catch(e) {
                console.log("All failed");
            } finally {
                document.getElementById('wait-modal').style.display = 'none';
            }
        }

        function updatePriceUI(price) {

            renderSimulation(true);
        }

        function connectWebSocket(symbol, name) {
            if(socket) socket.close(); 

            let wsUrl = "";
            if(apiMode === 'coinbase') {
                wsUrl = "wss://ws-feed.exchange.coinbase.com";
            } else if(apiMode === 'coincap') {
                wsUrl = `wss://ws.coincap.io/prices?assets=${name}`;
            } else {
                wsUrl = `wss://stream.binance.com:9443/ws/${symbol.toLowerCase()}usdt@ticker`;
            }

            socket = new WebSocket(wsUrl);

            socket.onopen = () => {
                if(apiMode === 'coinbase') {
                    const msg = { type: "subscribe", product_ids: [`${symbol}-USD`], channels: ["ticker"] };
                    socket.send(JSON.stringify(msg));
                }
            };

            socket.onerror = () => {
                if(apiMode === 'coinbase') { apiMode = 'coincap'; }
                else if(apiMode === 'coincap') { apiMode = 'binance'; }
                else { apiMode = 'coinbase'; }
                connectWebSocket(symbol, name);
            };

            socket.onmessage = function(event) {
                const data = JSON.parse(event.data);
                let price = 0;

                if(apiMode === 'coinbase' && data.type === 'ticker') {
                    price = parseFloat(data.price);
                } 
                else if(apiMode === 'coincap' && data[name]) {
                    price = parseFloat(data[name]);
                } 
                else if(apiMode === 'binance' && data.c) {
                    price = parseFloat(data.c);
                }

                if(price > 0) {
                    currentPrice = price;
                    document.getElementById('main-price').innerText = currentPrice.toFixed(2);
                    
                    if(apiMode === 'binance') {
                        const change = parseFloat(data.P);
                        const el = document.getElementById('price-change');
                        el.innerText = (change>0?'+':'') + change.toFixed(2) + '%';
                        el.style.color = change >= 0 ? 'var(--success-green)' : 'var(--danger-red)';
                    }
                    
                    renderSimulation(false);
                }
            };
        }

        function placeOrder(type) {
            if(currentPrice === 0) {
                document.getElementById('wait-modal').style.display = 'flex';
                raceForPrice();
                return;
            }
            const price = currentPrice.toFixed(2);
            if (type === 'Bullish') window.location.href = `mobile_bullish.html?symbol=${currentSym}&price=${price}`;
            if (type === 'Bearish') window.location.href = `mobile_bearish.html?symbol=${currentSym}&price=${price}`;
        }

        // SIMULATION - FIXED COLORS FOR QUANTITIES AND PRICES
        function renderSimulation(force) {
            if(currentPrice === 0) return;
            if(activeTab === 'order' || force) {
                let html = '';
                const spread = currentPrice * 0.0002;
                for(let i=0; i<8; i++) {
                    const bid = (currentPrice - (spread * (i+1))).toFixed(2);
                    const ask = (currentPrice + (spread * (i+1))).toFixed(2);
                    
                    html += `<div class="ob-row">
                        <span class="ob-qty" style="color:var(--text-light)">${(Math.random()*0.5).toFixed(4)}</span>
                        <span class="ob-price-bid text-green">${bid}</span>
                        <span class="ob-price-ask text-red">${ask}</span>
                        <span class="ob-qty-right" style="color:var(--text-light)">${(Math.random()*0.5).toFixed(4)}</span>
                    </div>`;
                }
                document.getElementById('ob-container').innerHTML = html;
            }
            if(Math.random() > 0.7 || force) addDealRow();
        }

        function addDealRow() {
            const isBuy = Math.random() > 0.5;
            const colorClass = isBuy ? 'text-green' : 'text-red';
            const typeText = isBuy ? 'Buy' : 'Sell';
            const time = new Date().toLocaleTimeString('en-US', {hour12:false, hour:"2-digit", minute:"2-digit", second:"2-digit"});
            const dealPrice = (currentPrice + (Math.random() - 0.5)).toFixed(2);
            const newRow = `<div class="deal-row"><span style="color:var(--text-muted)">${time}</span><span class="${colorClass}">${typeText}</span><span style="color:var(--text-light)">${dealPrice}</span><span style="text-align:right; color:var(--text-light)">${(Math.random()*0.5).toFixed(4)}</span></div>`;
            const container = document.getElementById('deal-container');
            container.insertAdjacentHTML('afterbegin', newRow);
            if(container.children.length > 20) container.removeChild(container.lastChild);
        }
